# ==================== 1. foundry 环境 (Python 3.12) ====================

# 用于: RFD3, MPNN, RF3 三个结构预测模型

cd /workspace
git clone https://gitee.com/gu-qianyi0424/foundry-production.git
cd foundry-production

conda create -n foundry python=3.12 -y
conda activate foundry

cd /workspace/foundry-production/foundry-production
pip install -e .

pip install -e ".[rf3]"
foundry install rfd3 ligandmpnn rf3 --checkpoint-dir ./checkpoints

rfd3 --help
mpnn --help
rf3 --help

# ==================== 2. binder 环境 (Python 3.10) ====================

# 用于: Streamlit 界面 + 热点预测(ML+DL) + 跨环境调度

conda create -n binder python=3.10 -y
conda activate binder
python --version # Python 3.10.x

# --- setuptools (pkg_resources 依赖) ---

pip install setuptools

# --- DGL ---

# ==================== 跨环境调用架构 ====================

#

# binder 环境 (Python 3.10) foundry 环境 (Python 3.12)

# ┌─────────────────────┐ ┌─────────────────────┐

# │ Streamlit 界面 │ │ RFD3 CLI │

# │ 热点预测 (ML+DL) │ │ MPNN CLI │

# │ │ conda_bridge │ RF3 CLI │

# │ rfd3_runner.py ────┼──────────────→ │ rfd3 design ... │

# │ mpnn_runner.py ────┼──────────────→ │ mpnn ... │

# │ rf3_runner.py ────┼──────────────→ │ rf3 fold ... │

# └─────────────────────┘ └─────────────────────┘

#

# conda_bridge.py 调用优先级:

# 1. conda run -n foundry (标准方式)

# 2. 直接使用 foundry 环境 Python 路径 (回退方式)

# 3. 降级为 mock 模式 (演示)

#

# 数据通过文件系统传递:

# binder 写入 PDB → foundry 读取 PDB → foundry 输出结果 → binder 读取结果

#
